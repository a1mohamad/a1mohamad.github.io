import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import MinMaxScaler
from sklearn import metrics
from sklearn.linear_model import LogisticRegression
from sklearn.svm import SVC
from xgboost import XGBClassifier

def evaluate_wine_models(data_path):
    '''
    Execute the complete supervised learning validation pipeline.
    Loads data, maps categorical boundaries, and evaluates model constraints.
    '''
    df = pd.read_csv(data_path)
    df['best quality'] = [1 if x > 5 else 0 for x in df.quality]

    # Preprocessing: Feature selection and scaling
    x = df.drop(['quality', 'best quality'], axis=1)
    y = df['best quality']
    scaler = MinMaxScaler()
    x_scaled = scaler.fit_transform(x)

    # Data Splitting
    xtrain, xtest, ytrain, ytest = train_test_split(x_scaled, y, test_size=0.2, random_state=42)

    # Define pipeline models
    models = [
        LogisticRegression(max_iter=1000), 
        XGBClassifier(eval_metric='logloss', n_estimators=100), 
        SVC(kernel='rbf', probability=True, C=1.0)
    ]

    # Iterate through classifiers
    for model in models:
        # Training Phase
        model.fit(xtrain, ytrain)
        
        # Inference Phase
        y_pred = model.predict(xtest)
        
        # Scoring Metrics
        auc_score = metrics.roc_auc_score(ytest, y_pred)
        accuracy = metrics.accuracy_score(ytest, y_pred)
        
        print(f"Model: {model.__class__.__name__}")
        print(f"Validation AUC: {auc_score:.4f}")
        print(f"Validation Accuracy: {accuracy:.4f}")
        print("-" * 30)
