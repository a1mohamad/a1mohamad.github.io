import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from xgboost import XGBClassifier

def execute_model_pipeline(data_path):
    '''
    Loads raw dataset, handles mean feature imputation, maps categories, and trains XGBoost classifier.
    '''
    # Extract features and targets matching experimental logic
    df = pd.read_csv(data_path)
    df['best quality'] = [1 if x > 5 else 0 for x in df.quality]
    
    features = df.drop(columns=['quality', 'best quality'])
    features = features.fillna(features.mean())
    target = df['best quality']
    
    # Execute deterministic stratified breakdown
    xtrain, xtest, ytrain, ytest = train_test_split(
        features, target, test_size=0.2, random_state=42
    )
    
    # Initialize optimal tree classifier engine
    model = XGBClassifier()
    model.fit(xtrain, ytrain)
    return model
